from django.core.management.base import BaseCommand
from django.utils import timezone
from datetime import timedelta
from frameworks.models import Control, Notification
from django.contrib.auth.models import User

class Command(BaseCommand):
    help = 'Processes recurring audit schedules and alerts contributors.'

    def handle(self, *args, **kwargs):
        today = timezone.now().date()
        target_date = today + timedelta(days=30) # Look 30 days ahead for upcoming audits

        # Find controls that have an upcoming audit and are currently passing/idle
        due_controls = Control.objects.filter(
            next_audit_date__lte=target_date,
            testing_frequency__in=["Monthly", "Quarterly", "Bi-Annual", "Annual"]
        ).exclude(status__in=["Ready for Audit", "Awaiting Approval", "Failing"])

        count = 0
        for control in due_controls:
            # 1. Reset Status to prompt Contributors for fresh evidence
            control.status = 'Not Evaluated'
            
            # 2. Advance the schedule to the next cycle
            if control.testing_frequency == 'Monthly':
                control.next_audit_date += timedelta(days=30)
            elif control.testing_frequency == 'Quarterly':
                control.next_audit_date += timedelta(days=90)
            elif control.testing_frequency == 'Bi-Annual':
                control.next_audit_date += timedelta(days=182)
            elif control.testing_frequency == 'Annual':
                control.next_audit_date += timedelta(days=365)
            
            control.save()

            # 3. Dispatch automated alerts to the people responsible for providing evidence
            users_to_notify = User.objects.filter(profile__role__in=['Contributor', 'Admin'],profile__notify_schedules=True)
            for u in users_to_notify:
                Notification.objects.create(
                    user=u,
                    title="Recurring Audit Cycle Starting",
                    description=f"Control {control.control_id} is approaching its scheduled {control.testing_frequency} audit. Please review and provide fresh evidence.",
                    severity="warning",
                    target_control_id=control.control_id
                )
            count += 1
        
        self.stdout.write(self.style.SUCCESS(f'Successfully initiated {count} new scheduled audit cycles.'))

def make_confidential_footer(company_name):
    def footer(canvas, doc):
        canvas.saveState()
        canvas.setFont('Helvetica-Bold', 9)
        canvas.setStrokeColorRGB(0.8, 0.8, 0.8)
        canvas.line(inch, 0.9 * inch, letter[0] - inch, 0.9 * inch)
        
        canvas.drawString(inch, 0.65 * inch, f"{company_name} | Generated: {timezone.now().strftime('%Y-%m-%d')}")
        canvas.setFillColor(colors.HexColor('#ef4444')) 
        canvas.drawCentredString(letter[0] / 2.0, 0.65 * inch, "CONFIDENTIAL - DO NOT DISTRIBUTE")
        canvas.setFillColor(colors.black)
        canvas.drawRightString(letter[0] - inch, 0.65 * inch, f"Page {doc.page}")
        canvas.restoreState()
    return footer

            canCopy=0, 
            canAnnotate=0
        )

        pdf_buffer,

class DRFTokenAuthMiddleware:
    """
    Reads the DRF Token from the headers and assigns the user to the request
    early in the request lifecycle so AuditlogMiddleware can see who it is.
    """
    def __init__(self, get_response):
        self.get_response = get_response

    def __call__(self, request):
        auth_header = request.META.get('HTTP_AUTHORIZATION', '')
        
        if auth_header.startswith('Token '):
            token_key = auth_header.split(' ')[1]
            try:
                token = Token.objects.get(key=token_key)
                request.user = token.user
            except Token.DoesNotExist:
                pass
                
        return self.get_response(request)

class Risk(models.Model):
    risk_id = models.CharField(max_length=50, unique=True)
    title = models.CharField(max_length=255)
    description = models.TextField(blank=True, null=True)
    inherent_likelihood = models.IntegerField(default=3)
    inherent_impact = models.IntegerField(default=3)
    mitigating_controls = models.ManyToManyField(Control, related_name='mitigated_risks', blank=True)

    @property
    def inherent_score(self):
        return self.inherent_likelihood * self.inherent_impact

    @property
    def residual_score(self):
        effective_count = 0
        
        mitigating_list = self.mitigating_controls.all()
        total_controls = len(mitigating_list)
        
        if total_controls == 0:
            return self.inherent_score

        for control in mitigating_list:
            if control.status == 'Passing' or (
                control.control_effectiveness == 'Effective' and 
                control.implementation_effectiveness == 'Effective'
            ):
                effective_count += 1
            elif control.status == 'Awaiting Approval':
                effective_count += 0.5

        mitigation_percentage = effective_count / total_controls
        reduction = self.inherent_score * (mitigation_percentage * 0.70)
        return max(1, round(self.inherent_score - reduction))

@receiver(post_save, sender=Control)
def auto_sync_crosswalked_controls(sender, instance, **kwargs):
    if getattr(instance, '_is_syncing', False) or getattr(instance, '_is_auto_linking', False):
        return
        
    siblings = instance.cross_walked_controls.all()
    if not siblings.exists():
        return
        
    instance._is_syncing = True
    orig = getattr(instance, '_original_state', {})
    
    status_changed = instance.status != orig.get('status')
    evidence_changed = instance.evidence != orig.get('evidence')
    effectiveness_changed = (
        instance.control_effectiveness != orig.get('control_effectiveness') or
        instance.implementation_effectiveness != orig.get('implementation_effectiveness')
    )
    
    eval_changed = status_changed or evidence_changed or effectiveness_changed
    
    if not eval_changed:
        cluster = [instance] + list(siblings)
        
        def get_eval_weight(c):
            w = 0
            if c.status in ["Passing", "Failing"]: w += 10
            elif c.status == "Awaiting Approval": w += 5
            elif c.status == "Ready for Audit": w += 3
            if c.evidence: w += 2
            if c.control_effectiveness != "Not Tested": w += 1
            if c.implementation_effectiveness != "Not Tested": w += 1
            return w
            
        source = max(cluster, key=get_eval_weight)
    else:
        source = instance
        
    for c in [instance] + list(siblings):
        if c.id == source.id:
            continue
            
        needs_save = False
        if c.status != source.status: c.status = source.status; needs_save = True
        if c.evidence != source.evidence: c.evidence = source.evidence; needs_save = True
        if c.control_effectiveness != source.control_effectiveness: c.control_effectiveness = source.control_effectiveness; needs_save = True
        if c.implementation_effectiveness != source.implementation_effectiveness: c.implementation_effectiveness = source.implementation_effectiveness; needs_save = True
        
        if needs_save:
            c._is_syncing = True
            c.save()
            c._is_syncing = False
            
    instance._original_state = {
        'status': instance.status,
        'evidence': instance.evidence,
        'control_effectiveness': instance.control_effectiveness,
        'implementation_effectiveness': instance.implementation_effectiveness,
        'tags': instance.tags
    }
    
    instance._is_syncing = False

from rest_framework.permissions import BasePermission, SAFE_METHODS

def get_user_role(request):
    """Helper to safely extract the user's role string."""
    if request.user.is_superuser:
        return 'Admin'
    if hasattr(request.user, 'profile'):
        return request.user.profile.role
    return 'Contributor' # Default fallback

class IsAdminUserOrReadOnly(BasePermission):
    def has_permission(self, request, view):
        if request.method in SAFE_METHODS:
            return True
        return get_user_role(request) == 'Admin'

class ControlMatrixPermissions(BasePermission):
    """
    - Admins: Full access.
    - Auditors & Reviewers: Can update status and evaluations, but cannot alter evidence.
    - Contributors: Can update evidence, but cannot set status to Passing/Awaiting.
    """
    def has_object_permission(self, request, view, obj):
        if request.method in SAFE_METHODS:
            return True
            
        role = get_user_role(request)
        if role == 'Admin':
            return True
            
        if role in ['Auditor', 'Reviewer']:
            req_evidence = request.data.get('evidence')
            if req_evidence and req_evidence != obj.evidence:
                return False
            return True
            
        if role == 'Contributor':
            req_status = request.data.get('status')
            if req_status in ['Passing', 'Not Applicable']:
                return False
            return True
            
        return False

class AssessmentPermissions(BasePermission):
    """
    - Admins: Full access.
    - Contributors: Read-only.
    - Auditors: Can create, and can only edit/delete if status is "In Progress".
    - Reviewers: Can only edit if status is "In QA Review".
    """
    def has_permission(self, request, view):
        if request.method in SAFE_METHODS:
            return True
        role = get_user_role(request)
        return role in ['Admin', 'Auditor', 'Reviewer']

    def has_object_permission(self, request, view, obj):
        if request.method in SAFE_METHODS:
            return True
            
        role = get_user_role(request)
        if role == 'Admin':
            return True
            
        if role == 'Auditor':
            if request.method == 'DELETE':
                return obj.status == 'In Progress'
            return obj.status == 'In Progress'
            
        if role == 'Reviewer':
            if request.method == 'DELETE':
                return False
            return obj.status == 'In QA Review'
            
        return False

class TestPlanPermissions(BasePermission):

class ControlSerializer(serializers.ModelSerializer):
    proofs = serializers.SerializerMethodField()

        many=True,
        read_only=True,
        slug_field='risk_id',
        source='mitigated_risks'
    )

    class Meta:
        model = Control

            'testing_frequency',
            'next_audit_date'

    def get_proofs(self, obj):
        proofs = {p.id: p for p in obj.proofs.all()}
        
        for sibling in obj.cross_walked_controls.all():
            for p in sibling.proofs.all():
                proofs[p.id] = p
                
        sorted_proofs = sorted(proofs.values(), key=lambda x: x.uploaded_at, reverse=True)
        return ProofSerializer(sorted_proofs, many=True, context=self.context).data

    def get_remediation_tasks(self, obj):
        tasks = {task.id: task for task in obj.remediation_tasks.all()}
        for sibling in obj.cross_walked_controls.all():
            for task in sibling.remediation_tasks.all():
                tasks[task.id] = task
        sorted_tasks = sorted(tasks.values(), key=lambda x: x.created_at, reverse=True)
        return RemediationTaskSerializer(sorted_tasks, many=True).data

    def update(self, instance, validated_data):
        request = self.context.get('request')
        user_role = getattr(request.user.profile, 'role', 'Contributor') if hasattr(request, 'user') and hasattr(request.user, 'profile') else 'Contributor'
        
        new_status = validated_data.get('status', instance.status)
        
        if user_role == 'Contributor' and new_status == 'Passing':
            raise serializers.ValidationError({"status": "Contributors cannot mark controls as Passing."})

        if new_status == 'Not Applicable' and user_role not in ['Admin', 'Contributor']:
            raise serializers.ValidationError({"status": "Only Admins and Contributors can mark controls as Not Applicable."})
            
        if user_role in ['Reviewer', 'Auditor']:
            if 'evidence' in validated_data and validated_data['evidence'] != instance.evidence:
                raise serializers.ValidationError({"evidence": "Reviewers and Auditors cannot modify text evidence."})
            validated_data.pop('description', None)
            
        return super().update(instance, validated_data)

class CustomAuthToken(ObtainAuthToken):
    def post(self, request, *args, **kwargs):
        serializer = self.serializer_class(data=request.data, context={'request': request})
        serializer.is_valid(raise_exception=True)
        user = serializer.validated_data['user']
        
        Token.objects.filter(user=user).delete()
        token = Token.objects.create(user=user)
        
        user_agent = request.META.get('HTTP_USER_AGENT', 'Unknown Browser')[:250]
        raw_ip = request.META.get('HTTP_X_FORWARDED_FOR', request.META.get('REMOTE_ADDR', '')).split(',')[0].strip()
        ip_address = raw_ip if raw_ip else None
        
        ActiveSession.objects.update_or_create(
            token_key=token.key,
            defaults={
                'user': user,
                'device_info': user_agent,
                'ip_address': ip_address,
                'location': "Appliance Network",
                'last_active': timezone.now()
            }
        )
        
        role = 'Contributor'
        avatar_url = None
        if hasattr(user, 'profile'):
            role = user.profile.role
            if user.profile.avatar:
                avatar_url = request.build_absolute_uri(user.profile.avatar.url)

        return Response({
            'token': token.key,
            'username': user.username,
            'role': role,
            'avatar': avatar_url
        })
    

class ActiveSessionsView(APIView):
    permission_classes = [IsAuthenticated]

    def get(self, request):
        user_role = getattr(request.user.profile, 'role', '') if hasattr(request.user, 'profile') else ''
        if user_role != 'Admin' and not request.user.is_superuser:
            return Response(
                {"error": "Only Administrators can view active sessions."}, 
                status=status.HTTP_403_FORBIDDEN
            )

        sessions = ActiveSession.objects.select_related('user').all().order_by('-last_active')
        data = []
        
        current_token = request.auth.key if hasattr(request, 'auth') and request.auth else None

        for s in sessions:
            data.append({
                'id': s.token_key,
                'device': f"[{s.user.username}] {s.device_info}", 
                'meta': s.ip_address or "Unknown IP",
                'location': s.location,
                'lastActive': s.last_active.strftime('%Y-%m-%d %H:%M'),
                'current': s.token_key == current_token
            })
            
        return Response(data)

    def delete(self, request):
        user_role = getattr(request.user.profile, 'role', '') if hasattr(request.user, 'profile') else ''
        if user_role != 'Admin' and not request.user.is_superuser:
            return Response(
                {"error": "Only Administrators can revoke sessions."}, 
                status=status.HTTP_403_FORBIDDEN
            )

        token_key = request.data.get('id')
        if not token_key:
            return Response({"error": "Session ID is required."}, status=status.HTTP_400_BAD_REQUEST)
        
        Token.objects.filter(key=token_key).delete()
        ActiveSession.objects.filter(token_key=token_key).delete()
        
        return Response({"message": "Session successfully revoked."}, status=status.HTTP_200_OK)


class ExportAuditPackageView(APIView):

        user_role = getattr(request.user.profile, 'role', '')
        if user_role not in ['Admin', 'Auditor', 'Executive']:
            return Response(
                {"error": "You do not have permission to export audit packages."}, 
                status=status.HTTP_403_FORBIDDEN
            )

        scope = request.data.get('scope', {})
        redact = request.data.get('redact', True)
        export_format = request.data.get('format', 'json')
        export_password = request.data.get('password')
        
        org_data = request.data.get('organization', {})
        company_name = org_data.get('companyName', 'Edge GRC')
        workspace_name = org_data.get('workspaceName', 'Compliance Solutions')
        logo_data_url = org_data.get('logoDataUrl')

        target_framework = request.data.get('target_framework', 'Generic Control Framework')
        assessment_boundary = request.data.get('assessment_boundary', workspace_name)
        
        export_data = {
            "metadata": {
                "generated_at": timezone.now().isoformat(),
                "generated_by": "REDACTED" if redact else request.user.username,
                "target_framework": target_framework,
                "assessment_boundary": assessment_boundary,
                "organization": company_name
            }
        }
        
        if scope.get('controls') or scope.get('evidence'):
            controls = Control.objects.prefetch_related(
                'proofs', 'remediation_tasks', 'cross_walked_controls', 
                'mitigated_risks', 'assessments'
            ).select_related('test_plan').all()
            
            controls_data = ControlSerializer(controls, many=True, context={'request': request}).data
            
            if not scope.get('evidence'):
                for c in controls_data:
                    c.pop('evidence', None)
                    c.pop('proofs', None)
                    
            export_data['controls'] = controls_data
            
        if scope.get('risks'):
            risks = Risk.objects.all().prefetch_related('mitigating_controls')
            risks_data = RiskSerializer(risks, many=True, context={'request': request}).data
            export_data['risks'] = risks_data

        if scope.get('logs'):
            logs = LogEntry.objects.select_related('actor').all().order_by('-timestamp')
            log_data = LogEntrySerializer(logs, many=True, context={'request': request}).data
            
            if redact:
                for log in log_data:
                    log['actor_name'] = 'REDACTED'
                    
            export_data['audit_logs'] = log_data

        if scope.get('summary'):
            total_controls = Control.objects.count()
            passing_controls = Control.objects.filter(status='Passing').count()
            failing_controls = Control.objects.filter(status='Failing').count()
            awaiting_controls = Control.objects.filter(status='Awaiting Approval').count()
            not_evaluated = Control.objects.filter(status='Not Evaluated').count()
            not_applicable = Control.objects.filter(status='Not Applicable').count()

            evaluatable_total = total_controls - not_applicable
            
            design_eff = Control.objects.filter(control_effectiveness='Effective').count()
            design_def = Control.objects.filter(control_effectiveness='Deficient').count()
            design_not = total_controls - (design_eff + design_def)
            
            op_eff = Control.objects.filter(implementation_effectiveness='Effective').count()
            op_def = Control.objects.filter(implementation_effectiveness='Deficient').count()
            op_not = total_controls - (op_eff + op_def)
            
            inherent_critical = inherent_high = inherent_medium = inherent_low = 0
            residual_critical = residual_high = residual_medium = residual_low = 0
            
            for risk in Risk.objects.all():
                i_score = risk.inherent_score
                if i_score >= 15: inherent_critical += 1
                elif i_score >= 10: inherent_high += 1
                elif i_score >= 5: inherent_medium += 1
                else: inherent_low += 1
                
                r_score = risk.residual_score
                if r_score >= 15: residual_critical += 1
                elif r_score >= 10: residual_high += 1
                elif r_score >= 5: residual_medium += 1
                else: residual_low += 1

            export_data['summary'] = {
                "total_controls": total_controls,
                "passing_controls": passing_controls,
                "failing_controls": failing_controls,
                "awaiting_controls": awaiting_controls,
                "not_evaluated": not_evaluated,
                "not_applicable": not_applicable,
                "compliance_score": round((passing_controls / evaluatable_total * 100) if evaluatable_total > 0 else 0),
                "design_effective": design_eff,
                "design_deficient": design_def,
                "design_not_tested": design_not,
                "operating_effective": op_eff,
                "operating_deficient": op_def,
                "operating_not_tested": op_not,
                "total_risks": Risk.objects.count(),
                "risk_inherent_critical": inherent_critical,
                "risk_inherent_high": inherent_high,
                "risk_inherent_medium": inherent_medium,
                "risk_inherent_low": inherent_low,
                "risk_residual_critical": residual_critical,
                "risk_residual_high": residual_high,
                "risk_residual_medium": residual_medium,
                "risk_residual_low": residual_low,
            }

        if export_format == 'pdf':
            pdf_buffer = generate_audit_pdf(export_data, company_name, export_password, logo_data_url)
            response = HttpResponse(pdf_buffer.getvalue(), content_type='application/pdf')
            response['Content-Disposition'] = 'attachment; filename="audit-report.pdf"'
            return response

        if export_format == 'zip':
            zip_buffer = io.BytesIO()
            with zipfile.ZipFile(zip_buffer, 'w', zipfile.ZIP_DEFLATED) as zip_file:
                zip_file.writestr('audit_summary.json', json.dumps(export_data, indent=2))
                
                if scope.get('evidence'):
                    # Pack standard Proof files
                    proofs = Proof.objects.all()
                    for proof in proofs:
                        if proof.file:
                            file_path = f"proofs/{proof.control.control_id}/{os.path.basename(proof.file.name)}"
                            try:
                                zip_file.writestr(file_path, proof.file.read())

                    # Pack the new Test Plan Attachments
                    test_plans = TestPlan.objects.exclude(attachment='')
                    for tp in test_plans:
                        if tp.attachment:
                            file_path = f"test_plans/{tp.control.control_id}/{os.path.basename(tp.attachment.name)}"
                            try:
                                zip_file.writestr(file_path, tp.attachment.read())
                            except Exception as e:
                                print(f"Failed to zip test plan file {file_path}: {e}")
                                
            response = HttpResponse(zip_buffer.getvalue(), content_type='application/zip')
            response['Content-Disposition'] = 'attachment; filename="audit-package.zip"'

            return Response({"error": "Invalid file format. Please provide a valid ZIP backup."}, status=status.HTTP_400_BAD_REQUEST)

            connection.close()
            
            with zipfile.ZipFile(file_obj, 'r') as zip_ref:
                file_names = zip_ref.namelist()
                db_path = settings.DATABASES['default']['NAME']
                media_root = settings.MEDIA_ROOT
                
                for name in file_names:
                    if name == 'db.sqlite3':

                            target_path = os.path.join(media_root, rel_path)
                            os.makedirs(os.path.dirname(target_path), exist_ok=True)
                            with open(target_path, 'wb') as media_file:
                                media_file.write(zip_ref.read(name))
                                
            return Response({"status": "System restored successfully."}, status=status.HTTP_200_OK)
            
        except Exception as e:
            return Response({"error": f"Failed to restore backup: {str(e)}"}, status=status.HTTP_500_INTERNAL_SERVER_ERROR)

def execute_system_update(zip_path, base_dir):
    try:
        with zipfile.ZipFile(zip_path, 'r') as zip_ref:
            zip_ref.extractall(base_dir)
            
        backend_dir = os.path.join(base_dir, 'grc-backend')
        manage_path = os.path.join(backend_dir, 'manage.py')
        if os.path.exists(manage_path):
            subprocess.run([sys.executable, manage_path, 'migrate'], cwd=backend_dir, check=True)
            
        subprocess.run(['sudo', 'systemctl', 'restart', 'grc-backend', 'grc-frontend'], check=True)
        
    except Exception as e:
        print(f"CRITICAL: System update failed: {e}")
    finally:
        if os.path.exists(zip_path):
            os.remove(zip_path)

class HTTPSConfigurationView(APIView):
    permission_classes = [IsAuthenticated]

    def get(self, request):
        if getattr(request.user.profile, 'role', '') != 'Admin':
            return Response({"error": "Admin access required."}, status=status.HTTP_403_FORBIDDEN)

        cert_path = os.path.join(settings.BASE_DIR, 'configs', 'ssl', 'appliance_cert.pem')
        
        if not os.path.exists(cert_path):
            return Response({"error": "Certificate has not been generated yet."}, status=status.HTTP_404_NOT_FOUND)

        with open(cert_path, 'rb') as f:
            response = HttpResponse(f.read(), content_type='application/x-x509-ca-cert')
            response['Content-Disposition'] = 'attachment; filename="edge-grc-root.pem"'
            return response

    def post(self, request):
        if getattr(request.user.profile, 'role', '') != 'Admin':
            return Response({"error": "Admin access required."}, status=status.HTTP_403_FORBIDDEN)

        # Extract values submitted from the frontend dialog
        common_name = request.data.get('common_name', 'edge-grc.local')
        org_name = request.data.get('organization', 'Edge-GRC Appliance')
        country = request.data.get('country', 'US')
        state = request.data.get('state', '')
        locality = request.data.get('locality', '')

        # 1. Generate Cryptographic Assets
        private_key = rsa.generate_private_key(public_exponent=65537, key_size=2048)
        
        # Build the dynamic X.509 Subject Name block
        attributes = [
            x509.NameAttribute(NameOID.COMMON_NAME, common_name),
            x509.NameAttribute(NameOID.ORGANIZATION_NAME, org_name),
        ]
        if country: 
            attributes.append(x509.NameAttribute(NameOID.COUNTRY_NAME, country))
        if state: 
            attributes.append(x509.NameAttribute(NameOID.STATE_OR_PROVINCE_NAME, state))
        if locality: 
            attributes.append(x509.NameAttribute(NameOID.LOCALITY_NAME, locality))

        subject = issuer = x509.Name(attributes)

        cert = x509.CertificateBuilder().subject_name(
            subject
        ).issuer_name(
            issuer
        ).public_key(
            private_key.public_key()
        ).serial_number(
            x509.random_serial_number()
        ).not_valid_before(
            timezone.now()
        ).not_valid_after(
            timezone.now() + datetime.timedelta(days=365)
        ).sign(private_key, hashes.SHA256())

        cert_pem = cert.public_bytes(serialization.Encoding.PEM)
        key_pem = private_key.private_bytes(
            encoding=serialization.Encoding.PEM,
            format=serialization.PrivateFormat.TraditionalOpenSSL,
            encryption_algorithm=serialization.NoEncryption()
        )

        # 2. Rewrite the Nginx Configuration File
        nginx_conf_path = os.path.join(settings.BASE_DIR, 'configs', 'nginx.conf')
        
        https_nginx_config = f"""server {{
    listen 80;
    server_name {common_name};
    
    # Force HTTP to redirect to HTTPS
    return 301 https://$host$request_uri;
}}

server {{
    listen 443 ssl;
    server_name {common_name};

    # Allow large payloads for system updates and database restores
    client_max_body_size 500M;

    ssl_certificate /opt/grc-appliance/grc-backend/configs/ssl/appliance_cert.pem;
    ssl_certificate_key /opt/grc-appliance/grc-backend/configs/ssl/appliance_key.pem;

    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;
    ssl_prefer_server_ciphers on;

    location /api/ {{
        proxy_pass http://127.0.0.1:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        
        # Increase timeouts to prevent 504 errors while Django processes 
        # large background tasks (e.g., extracting zips, running migrations)
        proxy_read_timeout 300s;
        proxy_connect_timeout 300s;
        proxy_send_timeout 300s;
    }}
    
    location /media/ {{
        alias /opt/grc-appliance/grc-backend/media/;
    }}

    location / {{
        proxy_pass http://127.0.0.1:3000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }}
}}
"""
        with open(nginx_conf_path, 'w') as f:
            f.write(https_nginx_config)

        # 3. Write Certificates to disk
        ssl_dir = os.path.join(settings.BASE_DIR, 'configs', 'ssl')
        os.makedirs(ssl_dir, exist_ok=True)

        with open(os.path.join(ssl_dir, 'appliance_key.pem'), 'wb') as f:
            f.write(key_pem)
            
        # Write the public cert LAST. This file modification triggers the Systemd 
        # 'grc-ssl-watcher.path' to execute 'systemctl reload nginx' immediately.
        with open(os.path.join(ssl_dir, 'appliance_cert.pem'), 'wb') as f:
            f.write(cert_pem)

        return Response({"status": "HTTPS successfully enabled. Nginx reloading."})
